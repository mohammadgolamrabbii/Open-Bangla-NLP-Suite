📦 Node.js / JavaScript SDKs

GitHub রিপো থেকে সরাসরি ব্যবহার বা `npm publish`-এর জন্য সম্পূর্ণ প্রস্তুত প্যাকেজসমূহ:

// ১. @open-bangla/normalizer
const { toBanglaCanonical, bijoyToUnicode } = require('@open-bangla/normalizer');
console.log(toBanglaCanonical("আমা\u09AF\u09BC")); // "আমায়" (Canonical NFC)
console.log(bijoyToUnicode("Avwg evsjv"));     // "আমি বাংলা"

// ২. @open-bangla/stemmer
const { BanglaMorphAnalyzer } = require('@open-bangla/stemmer');
const analyzer = new BanglaMorphAnalyzer();
console.log(analyzer.analyze("বইগুলোরই"));
// => { word: "বইগুলোরই", root: "বই", lemma: "বই", suffixes: [...] }

// ৩. @open-bangla/lexicon-core
const { BanglaLexiconIndex } = require('@open-bangla/lexicon-core');
const idx = new BanglaLexiconIndex(wordArray);
console.log(idx.suggest("আসির্বাদ")); // ["আশীর্বাদ"]
